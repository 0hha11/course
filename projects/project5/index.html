# 库存管理系统
import sqlite3

class InventoryManager:
    def __init__(self, db_name='inventory.db'):
        self.conn = sqlite3.connect(db_name)
        self.create_table()
    
    def create_table(self):
        """创建库存表"""
        cursor = self.conn.cursor()
        cursor.execute('''
            CREATE TABLE IF NOT EXISTS inventory (
                id INTEGER PRIMARY KEY,
                product_name TEXT,
                quantity INTEGER,
                min_stock INTEGER,
                price REAL,
                category TEXT
            )
        ''')
        self.conn.commit()
    
    def add_product(self, product_name, quantity, min_stock, price, category):
        """添加商品"""
        cursor = self.conn.cursor()
        cursor.execute('''
            INSERT INTO inventory (product_name, quantity, min_stock, price, category)
            VALUES (?, ?, ?, ?, ?)
        ''', (product_name, quantity, min_stock, price, category))
        self.conn.commit()
    
    def check_low_stock(self):
        """检查库存低于安全库存的商品"""
        cursor = self.conn.cursor()
        cursor.execute('SELECT * FROM inventory WHERE quantity < min_stock')
        return cursor.fetchall()

第2题 单选题

在SQL中，以下哪个语句用于更新数据？

正确答案：B

第3题 单选题

以下哪个不是SQLite的特点？

正确答案：C

SQLite是单文件数据库，不支持高并发写入。它的优点是轻量级、无需独立服务器、部署简单。
