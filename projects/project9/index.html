1. 异常值概述

异常值是指与其他数据点显著不同的值，可能影响分析结果的准确性。

import pandas as pd

df = pd.read_csv('retail_orders.csv')
print(df.head())

2. IQR方法

IQR（四分位数间距）方法是常用的异常值检测方法：

# 计算IQR
def detect_outliers_iqr(df, column):
    Q1 = df[column].quantile(0.25)
    Q3 = df[column].quantile(0.75)
    IQR = Q3 - Q1
    lower_bound = Q1 - 1.5 * IQR
    upper_bound = Q3 + 1.5 * IQR
    outliers = df[(df[column] < lower_bound) | (df[column] > upper_bound)]
    return outliers, lower_bound, upper_bound

# 检测异常值
outliers, lower, upper = detect_outliers_iqr(df, 'amount')
print(f"检测到{len(outliers)}个异常值")
print(f"下限: {lower:.2f}, 上限: {upper:.2f}")

3. Z-score方法

Z-score衡量数据点与均值的标准差距离：

# Z-score方法
from scipy import stats

# 计算Z-score
df['z_score'] = stats.zscore(df['amount'])

# 识别异常值（通常|Z| > 3）
outliers_z = df[abs(df['z_score']) > 3]
print(f"Z-score检测到{len(outliers_z)}个异常值")

4. 完整实战代码

import pandas as pd
import numpy as np
from scipy import stats
import matplotlib.pyplot as plt

plt.rcParams['font.sans-serif'] = ['SimHei']

# 读取数据
df = pd.read_csv('retail_orders.csv')

# 方法1: IQR方法
def detect_iqr(df, column):
    Q1 = df[column].quantile(0.25)
    Q3 = df[column].quantile(0.75)
    IQR = Q3 - Q1
    return df[(df[column] < Q1 - 1.5 * IQR) | (df[column] > Q3 + 1.5 * IQR)]

# 方法2: Z-score方法
def detect_zscore(df, column, threshold=3):
    df['z_score'] = stats.zscore(df[column])
    return df[abs(df['z_score']) > threshold]

# 检测销售额异常值
iqr_outliers = detect_iqr(df, 'total_amount')
z_outliers = detect_zscore(df, 'total_amount')

print(f"IQR方法检测到{len(iqr_outliers)}个异常值")
print(f"Z-score方法检测到{len(z_outliers)}个异常值")

# 可视化异常值
plt.figure(figsize=(12, 6))
plt.boxplot(df['total_amount'])
plt.title('销售额箱线图')
plt.show()

# 处理异常值：用中位数替换
median_amount = df['total_amount'].median()
df['total_amount_clean'] = np.where(
    df['total_amount'].isin(iqr_outliers['total_amount']),
    median_amount,
    df['total_amount']
)

print("\n处理后的统计描述:")
print(df[['total_amount', 'total_amount_clean']].describe())
