完整代码

import pandas as pd
import matplotlib.pyplot as plt

plt.rcParams['font.sans-serif'] = ['SimHei']
plt.rcParams['axes.unicode_minus'] = False

df = pd.read_csv('retail_orders.csv')

print("=== 数据清洗 ===")
df['order_date'] = pd.to_datetime(df['order_date'])
df['total_amount'] = df['price'] * df['quantity']
df['month'] = df['order_date'].dt.month
df['weekday'] = df['order_date'].dt.weekday

print("=== 探索性分析 ===")
print(f"总订单数: {len(df)}")
print(f"总销售额: {df['total_amount'].sum():.2f}")
print(f"平均订单金额: {df['total_amount'].mean():.2f}")

print("\n=== 类别分析 ===")
category_sales = df.groupby('category')['total_amount'].sum().sort_values(ascending=False)
print(category_sales)

print("\n=== 地区分析 ===")
region_sales = df.groupby('region')['total_amount'].sum().sort_values(ascending=False)
print(region_sales)

print("\n=== 月度趋势 ===")
monthly_sales = df.groupby('month')['total_amount'].sum()
print(monthly_sales)

print("\n=== 生成报告 ===")
report = {
    '总订单数': len(df),
    '总销售额': df['total_amount'].sum(),
    '平均订单金额': df['total_amount'].mean()
}
pd.DataFrame([report]).to_csv('sales_report.csv', index=False)
print("报告已生成！")
