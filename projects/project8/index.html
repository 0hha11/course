# 数据可视化仪表盘
import plotly.express as px
import plotly.graph_objects as go
from dash import Dash, dcc, html

app = Dash(__name__)

app.layout = html.Div([
    html.H1('数据可视化仪表盘'),
    
    # 数据卡片
    html.Div([
        html.Div('总销售额', className='card-title'),
        html.Div('¥1,234,567', className='card-value')
    ], className='stat-card'),
    
    # 图表
    dcc.Graph(
        figure=px.line(df, x='date', y='sales', title='销售趋势')
    ),
    
    dcc.Graph(
        figure=px.pie(df, values='amount', names='category', title='品类分布')
    )
])

if __name__ == '__main__':
    app.run_server(debug=True)
