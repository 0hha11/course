RFM模型概述

RFM模型是客户价值分析的经典方法，通过三个维度评估客户价值：

R (Recency) - 最近一次消费时间

距离上次消费的天数，越小越好

F (Frequency) - 消费频率

一段时间内的消费次数，越多越好

M (Monetary) - 消费金额

一段时间内的消费总额，越高越好

客户分群

根据RFM得分将客户分为不同群体。

r_labels = range(4, 0, -1)
f_labels = range(1, 5)
m_labels = range(1, 5)

rfm['R_score'] = pd.qcut(rfm['R'], q=4, labels=r_labels)
rfm['F_score'] = pd.qcut(rfm['F'], q=4, labels=f_labels)
rfm['M_score'] = pd.qcut(rfm['M'], q=4, labels=m_labels)

rfm['RFM_score'] = rfm['R_score'].astype(str) + rfm['F_score'].astype(str) + rfm['M_score'].astype(str)

def segment_customer(df):
    if df['RFM_score'] == '444':
        return '重要价值客户'
    elif df['R_score'] == '4':
        return '重要发展客户'
    elif df['F_score'] == '4':
        return '重要保持客户'
    elif df['M_score'] == '4':
        return '重要挽留客户'
    else:
        return '一般客户'

rfm['Segment'] = rfm.apply(segment_customer, axis=1)

print(rfm['Segment'].value_counts())

完整实战代码

import pandas as pd
import matplotlib.pyplot as plt

df = pd.read_csv('retail_orders.csv')
df['order_date'] = pd.to_datetime(df['order_date'])
df['total_amount'] = df['price'] * df['quantity']

reference_date = df['order_date'].max() + pd.Timedelta(days=1)

rfm = df.groupby('customer_id').agg({
    'order_date': lambda x: (reference_date - x.max()).days,
    'order_id': 'count',
    'total_amount': 'sum'
})
rfm.columns = ['R', 'F', 'M']

rfm['R_score'] = pd.qcut(rfm['R'], q=4, labels=range(4, 0, -1))
rfm['F_score'] = pd.qcut(rfm['F'], q=4, labels=range(1, 5), duplicates='drop')
rfm['M_score'] = pd.qcut(rfm['M'], q=4, labels=range(1, 5), duplicates='drop')

rfm['RFM_total'] = rfm['R_score'].astype(int) + rfm['F_score'].astype(int) + rfm['M_score'].astype(int)

def segment(score):
    if score >= 10: return '高价值客户'
    elif score >= 7: return '中价值客户'
    elif score >= 4: return '低价值客户'
    else: return '流失风险客户'

rfm['Segment'] = rfm['RFM_total'].apply(segment)

segment_counts = rfm['Segment'].value_counts()
print("客户分群结果:")
print(segment_counts)

rfm.to_csv('customer_segmentation.csv')
print("\n分析结果已保存！")
