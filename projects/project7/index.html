# 简单聊天机器人
import re

class ChatBot:
    def __init__(self):
        self.responses = {
            'hello': ['你好！', '您好！', '嗨！'],
            'bye': ['再见！', '下次见！', '拜拜！'],
            'name': ['我是聊天机器人小助手。', '你可以叫我小助手。']
        }
    
    def match_intent(self, message):
        """匹配用户意图"""
        message = message.lower()
        if re.search(r'你好|hello|hi', message):
            return 'hello'
        elif re.search(r'再见|bye|goodbye', message):
            return 'bye'
        elif re.search(r'名字|叫什么', message):
            return 'name'
        return None
    
    def respond(self, message):
        """生成响应"""
        intent = self.match_intent(message)
        if intent in self.responses:
            return self.responses[intent][0]
        return '我不太明白你的意思。'
