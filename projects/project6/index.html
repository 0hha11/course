# 用户行为分析
import pandas as pd

def analyze_user_behavior(df):
    """分析用户行为数据"""
    # 用户访问次数统计
    user_visits = df.groupby('user_id')['session_id'].nunique()
    
    # 平均停留时间
    df['duration'] = df['end_time'] - df['start_time']
    avg_duration = df.groupby('user_id')['duration'].mean()
    
    # 页面访问频次
    page_counts = df['page'].value_counts()
    
    return {'visits': user_visits, 'duration': avg_duration, 'pages': page_counts}

def funnel_analysis(df, funnel_steps):
    """漏斗分析"""
    funnel = {}
    for step in funnel_steps:
        users = df[df['page'] == step]['user_id'].unique()
        funnel[step] = len(users)
    return funnel
