1. 字符串基础操作

Pandas提供了丰富的字符串操作方法，通过.str访问器使用。

import pandas as pd

df = pd.DataFrame({'text': ['Hello World', 'Python Programming', 'Data Science']})

df['upper'] = df['text'].str.upper()
df['lower'] = df['text'].str.lower()
df['len'] = df['text'].str.len()

df['contains'] = df['text'].str.contains('Python')
df['startswith'] = df['text'].str.startswith('Hello')

print(df)

2. 字符串分割与提取

可以分割字符串或提取特定部分。

df['split'] = df['text'].str.split()
df['first_word'] = df['text'].str.split().str[0]

df['slice'] = df['text'].str[:5]

df['extract'] = df['text'].str.extract(r'(\w+)')

print(df[['text', 'first_word', 'slice']])

3. 正则表达式

正则表达式用于复杂的模式匹配和提取。

import pandas as pd
import re

text = "联系电话：13812345678，邮箱：test@example.com"

phones = re.findall(r'1[3-9]\d{9}', text)
emails = re.findall(r'\w+@\w+\.\w+', text)

print(f"电话: {phones}")
print(f"邮箱: {emails}")

df['digits'] = df['text'].str.replace(r'[^\d]', '', regex=True)

4. 文本清洗

文本清洗是文本分析的重要步骤。

df['cleaned'] = df['text'].str.strip()
df['cleaned'] = df['cleaned'].str.lower()
df['cleaned'] = df['cleaned'].str.replace(r'[^\w\s]', '', regex=True)
df['cleaned'] = df['cleaned'].str.replace(r'\s+', ' ', regex=True)

print(df[['text', 'cleaned']])

5. 完整实战代码

import pandas as pd
import re

df = pd.read_csv('customer_reviews.csv')

df['review_cleaned'] = df['review'].str.lower()
df['review_cleaned'] = df['review_cleaned'].str.replace(r'[^\w\s]', '', regex=True)
df['review_cleaned'] = df['review_cleaned'].str.strip()

df['word_count'] = df['review'].str.split().str.len()

df['has_positive'] = df['review'].str.contains('好|优秀|满意|推荐', regex=True)
df['has_negative'] = df['review'].str.contains('差|不好|失望|退货', regex=True)

def extract_rating(text):
    match = re.search(r'(\d)星|(\d)分', text)
    if match:
        return match.group(1) or match.group(2)
    return None

df['rating'] = df['review'].apply(extract_rating)

print(df[['review', 'word_count', 'has_positive', 'has_negative']].head())
