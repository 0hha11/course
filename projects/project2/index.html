# 销售数据分析
import pandas as pd
import matplotlib.pyplot as plt

def load_sales_data(filepath):
    """加载销售数据"""
    df = pd.read_csv(filepath)
    return df

def clean_data(df):
    """数据清洗"""
    df['date'] = pd.to_datetime(df['date'])
    df['total_amount'] = df['quantity'] * df['price']
    return df

def monthly_sales_trend(df):
    """月度销售趋势"""
    monthly = df.resample('M', on='date')['total_amount'].sum()
    monthly.plot(kind='line', figsize=(12, 6))
    plt.title('月度销售趋势')
    plt.xlabel('日期')
    plt.ylabel('销售额')
    plt.show()

def top_products(df, n=10):
    """TOP10产品"""
    top = df.groupby('product')['total_amount'].sum().sort_values(ascending=False).head(n)
    return top
